###Docker

Restart API:
docker compose -f infra\docker\docker-compose.dev.yml restart api

Rebuild API:
docker compose -f infra\docker\docker-compose.dev.yml up -d --build api
docker compose -f infra\docker\docker-compose.dev.yml up -d

Rebuild Web:
docker compose -f infra\docker\docker-compose.dev.yml up -d --build web
docker compose -f infra\docker\docker-compose.dev.yml up -d

Rebuild Both:
docker compose -f infra\docker\docker-compose.dev.yml up -d --build web api

Rebuild All: 
docker compose -f infra\docker\docker-compose.dev.yml down -v
docker compose -f infra/docker/docker-compose.dev.yml up -d --build

Docker Logs:
docker compose -f infra\docker\docker-compose.dev.yml logs -f --timestamps
